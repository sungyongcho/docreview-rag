`app/api/routes/stream.py`

```python
"""Server-sent-events streaming for the evidence-review workflow."""

from __future__ import annotations

import asyncio
from collections.abc import AsyncIterator
import contextlib
import json
from typing import Annotated

from fastapi import APIRouter, Depends
from fastapi.responses import StreamingResponse

from app.api.deps import ApiServices, get_api_services
from app.api.schemas import ReviewRequest, RunResponse, StreamNodeEvent
from app.observability import WorkflowNode
from app.workflow import WorkflowState

router = APIRouter(tags=["review"])
Services = Annotated[ApiServices, Depends(get_api_services)]

type _Event = tuple[str, str]


def _sse(event: str, data: str) -> str:
    """Format one server-sent event with a named type and single-line JSON data."""
    return f"event: {event}\ndata: {data}\n\n"


@router.post("/review/stream")
async def review_stream(request: ReviewRequest, services: Services) -> StreamingResponse:
    """Stream node completions while one guarded workflow runs, then its terminal run."""
    queue: asyncio.Queue[_Event | None] = asyncio.Queue()

    async def on_node(node: WorkflowNode, state: WorkflowState) -> None:
        event = StreamNodeEvent(
            node=node,
            evidence_count=len(state.evidence),
            relevant_count=len(state.relevant_chunk_ids),
            step_count=len(state.steps),
        )
        await queue.put(("node", event.model_dump_json()))

    async def run_review() -> None:
        try:
            report = await services.review_stream(request, on_node)
            payload = RunResponse.from_run_report(report).model_dump_json()
            await queue.put(("report", payload))
        except Exception as error:
            # str(error) can carry secrets (a SQLAlchemyError embeds the connection
            # string), so only the exception's class name leaves the process.
            payload = json.dumps({"error_type": type(error).__name__})
            await queue.put(("error", payload))
        finally:
            await queue.put(None)

    review_task = asyncio.create_task(run_review())

    async def events() -> AsyncIterator[str]:
        try:
            while (item := await queue.get()) is not None:
                yield _sse(*item)
            yield _sse("done", "{}")
        finally:
            # A disconnected client cancels the generator; stop the workflow with it.
            review_task.cancel()
            with contextlib.suppress(asyncio.CancelledError):
                await review_task

    return StreamingResponse(
        events(),
        media_type="text/event-stream",
        headers={"cache-control": "no-store"},
    )
```

[수정코드]

`app/api/routes/stream.py`

```python
"""Server-sent-events streaming for the evidence-review workflow."""

import asyncio
from collections.abc import AsyncGenerator, Mapping
import contextlib
import logging
from typing import Annotated

from fastapi import APIRouter, Depends
from fastapi.responses import StreamingResponse
from starlette.background import BackgroundTask
from starlette.types import Receive, Scope, Send

from app.api.deps import ApiServices, get_api_services
from app.api.errors import ApiProblemError
from app.api.schemas import ApiError, ErrorResponse, ReviewRequest, RunResponse, StreamNodeEvent
from app.observability import WorkflowNode
from app.workflow import WorkflowState

router = APIRouter(tags=["review"])
Services = Annotated[ApiServices, Depends(get_api_services)]

type _Event = tuple[str, str]

logger = logging.getLogger(__name__)


class _ClosingStreamingResponse(StreamingResponse):
    """Close the body iterator even when an ASGI send fails."""

    media_type = "text/event-stream"

    def __init__(
        self,
        content: AsyncGenerator[str],
        status_code: int = 200,
        headers: Mapping[str, str] | None = None,
        media_type: str | None = None,
        background: BackgroundTask | None = None,
    ) -> None:
        self._closable_content = content
        super().__init__(
            content,
            status_code=status_code,
            headers=headers,
            media_type=media_type or self.media_type,
            background=background,
        )

    async def __call__(self, scope: Scope, receive: Receive, send: Send) -> None:
        try:
            await super().__call__(scope, receive, send)
        finally:
            await self._closable_content.aclose()


def _sse(event: str, data: str) -> str:
    """Format one server-sent event with a named type and single-line JSON data."""
    return f"event: {event}\ndata: {data}\n\n"


@router.post(
    "/review/stream",
    response_class=_ClosingStreamingResponse,
    responses={
        200: {
            "description": "Server-sent workflow progress and terminal report events.",
            "content": {"text/event-stream": {"schema": {"type": "string"}}},
        }
    },
)
async def review_stream(request: ReviewRequest, services: Services) -> StreamingResponse:
    """Stream node progress and one secret-safe terminal result.

    Parameters
    ----------
    request : ReviewRequest
        Validated review request.
    services : Services
        Injected service boundary with observer support.

    Returns
    -------
    StreamingResponse
        SSE stream containing ``node`` events, one ``report`` or ``error``, and
        a final ``done`` marker.

    Notes
    -----
    The producer begins with iterator consumption and is cancelled when the body
    iterator closes, including ASGI send failures and client disconnects.
    """
    queue: asyncio.Queue[_Event | None] = asyncio.Queue()

    async def on_node(node: WorkflowNode, state: WorkflowState) -> None:
        event = StreamNodeEvent(
            node=node,
            evidence_count=len(state.evidence),
            relevant_count=len(state.relevant_chunk_ids),
            step_count=len(state.steps),
        )
        await queue.put(("node", event.model_dump_json()))

    async def run_review() -> None:
        try:
            report = await services.review_stream(request, on_node)
            payload = RunResponse.from_run_report(report).model_dump_json()
            await queue.put(("report", payload))
        except ApiProblemError as error:
            payload = ErrorResponse(error=error.error).model_dump_json()
            await queue.put(("error", payload))
        except Exception as error:
            logger.error(
                "Unhandled streamed review error",
                exc_info=(type(error), error, error.__traceback__),
            )
            payload = ErrorResponse(
                error=ApiError(
                    code="internal_error",
                    message="The request could not be completed.",
                )
            ).model_dump_json()
            await queue.put(("error", payload))
        finally:
            await queue.put(None)

    async def events() -> AsyncGenerator[str]:
        review_task = asyncio.create_task(run_review())
        try:
            while (item := await queue.get()) is not None:
                yield _sse(*item)
            yield _sse("done", "{}")
        finally:
            # A disconnected client cancels the generator; stop the workflow with it.
            review_task.cancel()
            with contextlib.suppress(asyncio.CancelledError):
                await review_task

    return _ClosingStreamingResponse(
        events(),
        headers={"cache-control": "no-store", "x-accel-buffering": "no"},
    )
```

[최적화내용]

1. producer task 수명과 `_ClosingStreamingResponse`

   - **변경**
     - review producer는 response 생성 시점이 아니라 body iterator가 실제 소비되기 시작할 때 생성한다.
     - response `__call__`의 `finally`가 body async generator를 명시적으로 닫아 ASGI 2.4 send 실패에서도 generator cleanup과 producer cancel을 실행한다.
   - **이유**
     - 기존 task는 iterator 진입 전 disconnect에서 아무도 취소하지 못했고, Starlette의 ASGI 2.4 send-error 경로는 body iterator를 자동 `aclose`하지 않았다.
   - **정합성**
     - 정상 FIFO와 done 순서를 유지하고, 첫 body send의 `OSError`가 `ClientDisconnect`로 끝날 때 blocking review task가 실제 `CancelledError`를 받는지 직접 ASGI 테스트했다.
   - **벤치마크**
     - 유료·장시간 작업의 누수를 막는 수명 수정이며 처리량 향상을 주장하지 않는다.

2. typed error event와 public redaction

   - **변경**
     - `ApiProblemError`는 기존 safe code/message의 `ErrorResponse`로, unexpected exception은 고정 `internal_error` envelope로 전송하고 서버에는 traceback을 기록한다.
     - terminal report는 public schema redaction과 runtime explicit-secret redaction을 지난 값만 직렬화한다.
   - **이유**
     - 기존 stream은 동기 503의 `provider_unavailable`을 `{"error_type":"ApiProblemError"}`로 축소했고, persistence만 편집한 채 원본 report를 SSE로 보낼 수 있었다.
   - **정합성**
     - budget/schema 같은 정상 structured failure는 계속 report event이고, typed service failure·unexpected failure·secret sentinel을 분리해 검증했다.
   - **벤치마크**
     - 안전성·계약 수정이며 성능 향상을 주장하지 않는다.

3. SSE OpenAPI와 proxy header

   - **변경**
     - response class의 기본 media type을 `text/event-stream`으로 선언하고 OpenAPI 200 content를 같은 media type으로 기록한다.
     - `cache-control: no-store`와 `x-accel-buffering: no`를 함께 보낸다.
   - **이유**
     - 실제 wire format은 SSE인데 OpenAPI는 `application/json`을 광고했고 reverse proxy buffering은 진행 이벤트 전달을 늦출 수 있다.
   - **정합성**
     - 기존 TestClient content type과 event parser가 유지되며 OpenAPI media type을 별도 단언한다.
   - **벤치마크**
     - 전송 계약 수정이며 성능 향상을 주장하지 않는다.

4. heartbeat: 없음

   - **변경**
     - 주기적 comment heartbeat는 추가하지 않았다.
   - **이유**
     - 필요한 간격은 배포 proxy timeout 정책에 의존하며 M7 배포 설정 없이 임의 주기를 넣으면 불필요한 wakeup과 테스트 시간을 만든다.
   - **정합성**
     - 현재 최대 이벤트 수가 node 4개와 terminal·done으로 제한되어 queue 메모리는 bounded behavior다.
   - **벤치마크**
     - 측정값이 없어 성능 향상을 주장하지 않는다.
