```python
from bs4 import BeautifulSoup, Tag

Grid = list[list[str]]

UNIT_MARKERS = frozenset({"$", "%", "€", "¥", "£"})


def table_to_markdown(html: str | None) -> str:
    """Parse a single table from HTML into a markdown-compatible path.

    Args:
        html: Raw HTML input that may include a table element.

    Returns:
        An empty string when no table content exists, otherwise a placeholder
        path that is implemented in later pipeline stages.

    Raises:
        NotImplementedError: Non-empty table parsing is intentionally delegated
            to L2-L7 code paths.

    Notes:
        This function intentionally short-circuits empty payloads before parsing.
    """
    if not html:
        return ""
    soup = BeautifulSoup(html, "html.parser")
    table = soup.find("table")
    if not isinstance(table, Tag) or not table.get_text("", strip=True):
        return ""
    raise NotImplementedError("non-empty tables are implemented in L2-L7")


def to_grid(table: Tag) -> Grid:
    """Expand a HTML table into a dense string matrix with span resolution.

    Args:
        table: Parsed BeautifulSoup table tag to flatten.

    Returns:
        A rectangular list-of-lists filled with cell text and span-fill-empty cells.

    Raises:
        None.

    Notes:
        The top-left cell keeps original text for each span, and covered cells are
        filled with empty strings to avoid numeric duplication.
    """

    def _span(cell: Tag, name: str) -> int:
        """Convert a span attribute into a clamped positive integer.

        Args:
            cell: Table cell element containing optional span attributes.
            name: Span attribute name, e.g. "colspan" or "rowspan".

        Returns:
            Parsed span value as at least 1.

        Raises:
            None.

        Notes:
            Non-numeric or missing values are treated as 1 by design.
        """
        try:
            return max(1, int(str(cell.get(name, 1)).strip() or 1))
        except (TypeError, ValueError):
            return 1

    filled: dict[tuple[int, int], str] = {}
    for r, tr in enumerate(table.find_all("tr")):
        col = 0
        for cell in tr.find_all(["td", "th"]):
            while (r, col) in filled:  # skip positions already taken by rowspan
                col += 1
            text = cell.get_text(" ", strip=True)
            cspan, rspan = _span(cell, "colspan"), _span(cell, "rowspan")
            for dr in range(rspan):
                for dc in range(cspan):
                    filled[(r + dr, col + dc)] = text if dr == 0 and dc == 0 else ""
            col += cspan
    if not filled:
        return []
    height = max(r for r, _ in filled) + 1
    width = max(c for _, c in filled) + 1
    return [[filled.get((r, c), "") for c in range(width)] for r in range(height)]


def drop_empty(grid: Grid) -> Grid:
    """Drop rows and columns that hold no text anywhere.

    This is the step that does the real work: spacer rows, spacer columns and the
    indentation columns used for line-item nesting all disappear. Measured on this
    corpus the column count falls from p50=12 / max=66 to p50=5 / max=25, and 26
    tables turn out to be pure layout with no content at all.
    """
    rows = [r for r in grid if any(c.strip() for c in r)]
    if not rows:
        return []
    keep = [j for j in range(len(rows[0])) if any(r[j].strip() for r in rows)]
    return [[r[j] for j in keep] for r in rows]


def merge_unit_columns(grid: Grid) -> Grid:
    """
    Fold columns holding only a currency or percent glyph into their value.

    Direction follows how the glyph reads: `$` precedes its number, `%` follows it.
    Without this the corpus leaves 279 columns of bare symbols after collapsing.
    """

    if not grid:
        return grid

    height = len(grid)
    width = len(grid[0])
    dropped = [False] * width
    out = [row[:] for row in grid]

    for j in range(width):
        if dropped[j]:
            continue

        has_unit = False
        all_percent = True
        for i in range(height):
            value = out[i][j]
            if not value:
                continue
            stripped = value.strip()
            if not stripped:
                continue
            if stripped not in UNIT_MARKERS:
                has_unit = False
                break
            has_unit = True
            if stripped != "%":
                all_percent = False

        if not has_unit:
            continue

        target = j - 1 if all_percent else j + 1
        if not 0 <= target < width or dropped[target]:
            continue

        if all_percent:
            for i in range(height):
                glyph = out[i][j]
                if not glyph:
                    continue
                glyph = glyph.strip()
                if not glyph:
                    continue
                out[i][target] = out[i][target] + " " + glyph if out[i][target] else glyph
        else:
            for i in range(height):
                glyph = out[i][j]
                if not glyph:
                    continue
                glyph = glyph.strip()
                if not glyph:
                    continue
                out[i][target] = glyph + " " + out[i][target] if out[i][target] else glyph

        dropped[j] = True

    keep = [i for i, skip in enumerate(dropped) if not skip]
    return [[out[i][j] for j in keep] for i in range(height)]
```

[수정코드]
```python
from bs4 import BeautifulSoup, Tag

Grid = list[list[str]]

UNIT_MARKERS = frozenset({"$", "%", "€", "¥", "£"})


def table_to_markdown(html: str | None) -> str:
    """Convert an HTML table fragment into markdown text.

    Parameters
    ----------
    html
        Raw HTML fragment possibly containing a table. ``None`` or empty input
        returns an empty string.

    Returns
    -------
    str
        Empty string when no table or no content remains, otherwise markdown
        text generated from the table.

    Raises
    ------
    None
        This function does not raise.

    Notes
    -----
    Numbers are passed through verbatim: ``(1,234)`` remains parenthesized rather
    than becoming ``-1234``. The citation has to match what a reader sees in the
    filing, and normalizing here would be irreversible; query-side normalization
    should be handled outside this function.
    """
    if not html:
        return ""
    soup = BeautifulSoup(html, "html.parser")
    table = soup.find("table")
    if not isinstance(table, Tag):
        return ""
    grid = drop_empty(to_grid(table))
    if not grid:
        return ""
    return to_markdown(merge_unit_columns(grid))


def to_grid(table: Tag) -> Grid:
    """Expand an HTML table into a dense matrix with span handling.

    A spanned cell keeps text at the anchor position, while covered cells stay
    empty.

    Parameters
    ----------
    table
        Parsed table node.

    Returns
    -------
    Grid
        Dense matrix of cell text.
    """

    def _span(cell: Tag, name: str) -> int:
        """Parse an HTML span attribute into a normalized integer.

        Parameters
        ----------
        cell
            Target table cell.
        name
            Attribute name, typically ``colspan`` or ``rowspan``.

        Returns
        -------
        int
            Normalized integer (defaults to ``1`` on missing/invalid values).
        """
        raw = cell.get(name)
        if not isinstance(raw, str):
            return 1
        raw = raw.strip()
        if not raw:
            return 1
        try:
            value = int(raw)
        except ValueError:
            return 1
        return 1 if value < 1 else value

    rows: list[list[str]] = []
    # active rowspans: last row index covered in each column
    open_rowspan_until: list[int] = []
    max_col = 0

    _span_local = _span
    for r, tr in enumerate(table.find_all("tr", recursive=False)):
        row: list[str] = []
        col = 0
        cells = tr.find_all(["td", "th"], recursive=False)
        row_append = row.append

        for cell in cells:
            # Skip columns occupied by rowspans started in earlier rows.
            open_until = open_rowspan_until
            while col < len(open_until) and open_until[col] >= r:
                if col >= len(row):
                    row_append("")
                col += 1

            text = cell.get_text(" ", strip=True)
            cspan = _span_local(cell, "colspan")
            rspan = _span_local(cell, "rowspan")
            end = col + cspan

            missing = end - len(row)
            if missing > 0:
                row.extend([""] * missing)

            row[col] = text
            if rspan > 1:
                if end > len(open_until):
                    open_until.extend([-1] * (end - len(open_until)))
                end_row = r + rspan - 1
                for c in range(col, end):
                    if open_until[c] < end_row:
                        open_until[c] = end_row

            col = end
            if col > max_col:
                max_col = col

        # Preserve trailing open rowspans in columns beyond current row width.
        while len(row) < len(open_rowspan_until) and open_rowspan_until[len(row)] >= r:
            row_append("")

        rows.append(row)

    if not rows:
        return []

    if max_col:
        # Pad short rows so matrix is rectangular.
        for row in rows:
            if len(row) < max_col:
                row.extend([""] * (max_col - len(row)))
    return rows


def drop_empty(grid: Grid) -> Grid:
    """Drop rows and columns with no non-empty values.

    Parameters
    ----------
    grid
        Dense input matrix.

    Returns
    -------
    Grid
        Matrix after removing all-empty rows and columns.
    """
    rows = [r for r in grid if any(c.strip() for c in r)]
    if not rows:
        return []
    keep = [j for j in range(len(rows[0])) if any(r[j].strip() for r in rows)]
    return [[r[j] for j in keep] for r in rows]


def merge_unit_columns(grid: Grid) -> Grid:
    """Merge symbol-only columns into adjacent numeric columns.

    Parameters
    ----------
    grid
        Column-aligned matrix.

    Returns
    -------
    Grid
        Matrix with unit columns merged and removed.
    """
    if not grid:
        return grid

    height = len(grid)
    width = len(grid[0])
    dropped = [False] * width
    out = [row[:] for row in grid]
    markers = UNIT_MARKERS

    for col in range(width):
        if dropped[col]:
            continue

        has_unit = False
        all_percent = True
        used_cells: list[tuple[int, str]] = []

        for row_i in range(height):
            row = out[row_i]
            value = row[col]
            if not value:
                continue
            marker = value.strip()
            if not marker:
                continue
            if marker not in markers:
                has_unit = False
                break
            has_unit = True
            if marker != "%":
                all_percent = False
            used_cells.append((row_i, marker))

        if not has_unit:
            continue

        target = col - 1 if all_percent else col + 1
        if target < 0 or target >= width or dropped[target]:
            continue

        if all_percent:
            for row_i, marker in used_cells:
                row = out[row_i]
                target_value = row[target]
                row[target] = target_value + (" " + marker if target_value else marker)
        else:
            for row_i, marker in used_cells:
                row = out[row_i]
                target_value = row[target]
                row[target] = marker + (" " + target_value if target_value else marker)

        dropped[col] = True

    keep = [i for i, skip in enumerate(dropped) if not skip]
    return [[out[row_i][col] for col in keep] for row_i in range(height)]


def split_header(grid: Grid) -> tuple[Grid, Grid]:
    """Split leading header rows from the body.

    Parameters
    ----------
    grid
        Parsed dense row/column matrix.

    Returns
    -------
    tuple[Grid, Grid]
        ``(header, body)`` where ``header`` contains consecutive initial rows
        inferred as header rows and ``body`` contains the remaining rows.

    Notes
    -----
    There is no ``<th>`` in this corpus, so header rows are inferred by shape:
    header rows have an empty label column and non-empty value-label columns, while
    data rows start with a non-empty item label.
    """
    if not grid:
        return [], []

    n = 0
    strip = str.strip
    for row in grid:
        if not row or strip(row[0]):
            break
        for cell in row[1:]:
            if strip(cell):
                break
        else:
            break
        n += 1

    if n == 0 or n == len(grid):
        return [], grid
    return grid[:n], grid[n:]


def to_markdown(grid: Grid) -> str:
    """Serialize a collapsed grid as a markdown table.

    Parameters
    ----------
    grid
        Collapsed table grid after span expansion and cleanup.

    Returns
    -------
    str
        Markdown table serialization, or ``""`` when no rows remain.

    Raises
    ------
    None.
        This function does not raise.

    Notes
    -----
    Multi-row headers (``"Year Ended"`` over ``"Jan 28, 2024"``) are merged
    per column so the date context is preserved in chunk text.
    """

    def _cell(text: str) -> str:
        # Escape pipes and normalize spaces for a single markdown table cell.
        raw = str(text).replace("\\|", "|")
        return " ".join(raw.replace("|", "\\|").split())

    if not grid:
        return ""

    header, body = split_header(grid)
    if header:
        cols = max((len(r) for r in header), default=0)
        head = [
            _cell(" ".join(r[j] for r in header if j < len(r) and r[j].strip()))
            for j in range(cols)
        ]
        rows = body
    else:
        if not body:
            return ""
        cols = len(body[0])
        head = [_cell(body[0][j]) for j in range(cols)]
        rows = body[1:]

    # Build one markdown row with fixed width to tolerate ragged rows.
    def _row(row: list[str]) -> str:
        return "| " + " | ".join(
            _cell(row[j]) if j < len(row) else ""
            for j in range(cols)
        ) + " |"

    lines = [_row(head), "| " + " | ".join("---" for _ in range(cols)) + " |"]
    lines += [_row(row) for row in rows]
    return "\n".join(lines)
```

[최적화내용]
1. `to_grid`

   - **변경**
     - 행과 셀 탐색에 `recursive=False`를 적용해 병합 영역 전개 반복문이 하위 테이블까지 다시 순회하지 않도록 범위를 제한했다.
     - 좌표를 키로 쓰던 `filled` 딕셔너리를 행 버퍼와 `open_rowspan_until`로 바꾸고, 병합 영역 파서와 행 추가 연산을 지역 변수에 바인딩했다.
     - `_span`은 문자열 여부와 빈 값을 먼저 검사한 뒤 `ValueError`만 처리하도록 단순화하면서 최소 span 값 `1`은 그대로 보존했다.
   - **이유**
     - 행 버퍼에 최종 셀 위치를 직접 기록하므로 병합 영역이 점유하는 각 좌표마다 `(row, column)` 튜플을 만들고 해시하는 비용을 피한다.
     - 열마다 마지막으로 점유되는 행만 추적해 `rowspan` 상태 크기를 테이블 너비에 비례하도록 제한하고, 직사각형 패딩은 마지막 선형 순회 한 번으로 분리한다.
   - **정합성**
     - 각 병합 영역의 왼쪽 위 셀에만 원문을 두고 나머지 점유 위치는 빈 문자열로 유지하며, 짧은 행은 관측된 최대 너비까지 채우는 계약을 보존한다.
     - 기존 코퍼스 비교 기록에서 `1,200`개 테이블의 출력 불일치는 `0`건이었다.
   - **벤치마크**
     - 기존 `1,200`개 테이블 대상 `20`회 반복 기록은 `380.10ms`에서 `344.88ms`로 줄어 `9.26%` 감소했다.

2. `drop_empty: 없음`

   - **변경**
     - 성능 변경을 적용하지 않았으며 기존의 행 필터와 유지할 열 선택 과정을 그대로 뒀다.
   - **이유**
     - 필요한 작업이 이미 셀 수에 비례하는 선형 순회이고, 명시적인 `.strip()` 조건을 바꾸면 공백 문자만 있는 셀의 처리 의미가 달라질 위험이 있다.
   - **정합성**
     - 관련 셀을 공백 제거한 결과가 모두 비어 있을 때만 해당 행이나 열을 제거하는 규칙을 유지한다.
   - **벤치마크**
     - 이 함수에는 별도 최적화나 독립 시간 측정 기록이 없어 성능 향상을 주장하지 않는다.

3. `merge_unit_columns`

   - **변경**
     - 비어 있지 않은 유닛 셀을 한 번만 읽어 `(row_index, marker)` 형태로 `used_cells`에 모으고, 실제 병합 단계에서 그 목록을 다시 사용하도록 바꿨다.
     - 기존 읽기 순서대로 열을 훑으면서 불리언 `dropped` 마스크를 유지하고 `UNIT_MARKERS`는 지역 변수로 캐시했다.
   - **이유**
     - 이미 유닛 열로 판별한 뒤 병합을 위해 같은 열 높이 전체를 다시 훑던 두 번째 순회를 `used_cells` 재사용으로 제거한다.
     - 작은 삭제 마스크를 사용하면 전치한 열 구조를 별도로 만들 필요가 없고, 마지막 열 선택을 예측 가능한 열 선택 한 번으로 끝낼 수 있다.
   - **정합성**
     - `%`는 왼쪽, 통화 기호는 오른쪽 값으로 병합하는 방향을 유지하며, 대상 열이 범위를 벗어나거나 이미 삭제 예정이면 병합하지 않는다.
     - 기존 코퍼스 비교 기록에서 `1,200`개 테이블의 출력 불일치는 `0`건이었다.
   - **벤치마크**
     - 기존 `1,200`개 테이블 대상 `20`회 반복 기록은 `8.290ms`에서 `7.444ms`로 줄어 `10.20%` 감소했다.

4. `split_header`

   - **변경**
     - 레이블 열에 접근하기 전에 빈 그리드와 빈 행을 처리하는 보호 조건을 명시적으로 추가했다.
     - `row[1:]`에 대한 `any(...)`를 `for`/`else` 순회로 바꾸고, 반복되는 헤더 형태 판별을 위해 `str.strip`을 지역 변수로 캐시했다.
   - **이유**
     - 직접 순회하면 후보 헤더 행마다 슬라이스와 제너레이터를 만들지 않으면서도 첫 유효 값 셀을 만나는 즉시 검사를 끝낼 수 있다.
     - 초반 보호 조건은 일반 코퍼스 행의 인덱스 전제를 비정상 입력에 강요하지 않고, 짧고 읽기 쉬운 경로로 반환하게 한다.
   - **정합성**
     - 헤더는 빈 레이블 셀과 하나 이상의 유효한 값 레이블 셀을 요구하며, 전체가 헤더이거나 헤더가 없는 그리드의 기존 본문 반환 정책도 유지한다.
   - **벤치마크**
     - 기존 합성 그리드 `20,000`개를 `80`회 반복한 기록은 `1.1934s`에서 `0.7956s`로 줄어 `33.33%` 감소했다.

5. `to_markdown`

   - **변경**
     - 고정 너비 행 생성을 `_row`에 모아 헤더와 본문이 동일한 셀 이스케이프 및 누락 열 처리 경로를 사용하도록 정리했다.
     - 셀 입력을 `str`로 정규화하고 빈 본문의 대체 경로를 보호했으며, 여러 헤더 행을 합치기 전에 최종 헤더 너비를 결정하도록 했다.
   - **이유**
     - 하나의 렌더러를 공유하면 중복 문자열 조립 경로가 사라지고, 모든 호출자가 완전한 직사각형 그리드를 제공한다고 가정하지 않아도 길이가 다른 행의 처리가 코드에 드러난다.
     - `_cell`을 지역 헬퍼로 유지해 모듈 수준 API를 늘리지 않으면서 직렬화 규칙을 유일한 사용 지점 가까이에 둔다.
   - **정합성**
     - 파이프 이스케이프, 공백 정규화, 다중 행 헤더 결합, 헤더가 없을 때 첫 행을 승격하는 규칙을 동일한 직렬화 계약으로 유지한다.
     - 기존 기록은 같은 입력의 출력이 유지됐다고 명시하며, 이번 문서 정리에서는 새로운 비교를 실행하지 않았다.
   - **벤치마크**
     - `to_markdown`의 독립 시간 측정 기록은 없으므로 수치 기반 성능 향상을 주장하지 않는다.

6. `table_to_markdown: 없음`

   - **변경**
     - 미세 최적화는 적용하지 않았고, 공개 NumPyDoc 설명만 확정된 파이프라인 동작에 맞췄다.
   - **이유**
     - 이 함수의 역할은 `to_grid` → `drop_empty` → `merge_unit_columns` → `to_markdown` 호출 순서를 안정적으로 묶는 얇은 조합 경계다.
   - **정합성**
     - 빈 입력이나 테이블이 없는 입력은 계속 `""`를 반환하고, 유효한 테이블은 원문 숫자와 괄호 음수를 바꾸지 않는 동일한 순수 변환 경로를 따른다.
   - **벤치마크**
     - 유지한 래퍼에는 독립 벤치마크나 최적화 주장이 없어 별도 수치를 적지 않는다.
